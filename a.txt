function FormatTime {
    param($Time)

}

function Emit {
    param([System.Collections.Generic.List[string]]$Data, [string]$Text, [string]$Color = 'Gray')
    foreach ($Line in ($Text -split "`r?`n")) {
        $Data.Add($Line)
        Write-Host $Line -ForegroundColor $Color
    }
}

function PushToDiscord {
    param ($Content, $Name, $CustomHeader)
    if ($Webhook -like '*PASTE_*') {
        Write-Host "[!] No webhook set at the top of the script - report was NOT sent." -ForegroundColor Yellow
        return
    }
    try {
        $Latin1 = [System.Text.Encoding]::GetEncoding("iso-8859-1")
        $ToLatin = { param($s) $Latin1.GetString([System.Text.Encoding]::UTF8.GetBytes($s)) }

        if ($CustomHeader.Length -gt 1900) { $CustomHeader = $CustomHeader.Substring(0, 1900) }
        $Payload  = & $ToLatin $Content
        $Header   = & $ToLatin $CustomHeader
        $Boundary = [System.Guid]::NewGuid().ToString()
        $LineBreak = "`r`n"

        $FormConfig = (
            "--$Boundary",
            "Content-Disposition: form-data; name=`"content`"",
            "",
            $Header,
            "--$Boundary",
            "Content-Disposition: form-data; name=`"file`"; filename=`"$Name`"",
            "Content-Type: text/plain; charset=utf-8",
            "",
            $Payload,
            "--$Boundary--"
        )
        $DataBuffer = $Latin1.GetBytes(($FormConfig -join $LineBreak))
        Invoke-RestMethod -Uri $Webhook -Method Post -ContentType "multipart/form-data; boundary=$Boundary" -Body $DataBuffer | Out-Null
        Write-Host "[+] Report sent to Discord." -ForegroundColor DarkGreen
    } catch {
        Write-Host "[!] Discord upload failed: $($_.Exception.Message)" -ForegroundColor Red
    }
}

# ------------------------------------------------------------
#  Drive detection
# ------------------------------------------------------------
function Get-UsbVolumes {
    # Removable drives (type 2) PLUS fixed-type drives that sit on a USB bus (USB SSDs / enclosures)
    $UsbFixed = @()
    try {
        $UsbFixed = @(Get-Disk -ErrorAction Stop | Where-Object { $_.BusType -eq 'USB' } |
            Get-Partition -ErrorAction SilentlyContinue |
            Where-Object { $_.DriveLetter -match '[A-Za-z]' } |
            ForEach-Object { "$($_.DriveLetter):".ToUpper() })
    } catch {}
    Get-CimInstance Win32_LogicalDisk | Where-Object {
        $_.DriveType -eq 2 -or ($_.DriveType -eq 3 -and $UsbFixed -contains $_.DeviceID.ToUpper())
    }
}

function Get-InternalLetters {
    $Usb = @(Get-UsbVolumes | ForEach-Object { $_.DeviceID.ToUpper() })
    @(Get-CimInstance Win32_LogicalDisk | Where-Object {
        $_.DriveType -in 3, 4, 5 -and $Usb -notcontains $_.DeviceID.ToUpper()
    } | ForEach-Object { $_.DeviceID.ToUpper() })
}

function Get-VolumeMap {
    # \Device\HarddiskVolumeN -> X:
    $Map = @{}
    foreach ($L in [char[]](65..90)) {
        $Sb = New-Object System.Text.StringBuilder 520
        if ([DosDev]::QueryDosDevice("$($L):", $Sb, 520) -gt 0) { $Map[$Sb.ToString()] = "$($L):" }
    }
    $Map
}

function Test-ExternalPath {
    param([string]$Path, [string[]]$Internal)
    if ($Path -notmatch '^([A-Za-z]:)\\') { return $false }
    return ($Internal -notcontains $Matches[1].ToUpper())
}

function Invoke-Rot13 {
    param([string]$Text)
    -join ($Text.ToCharArray() | ForEach-Object {
        $c = [int]$_
        if ($c -ge 65 -and $c -le 90)      { [char](($c - 65 + 13) % 26 + 65) }
        elseif ($c -ge 97 -and $c -le 122) { [char](($c - 97 + 13) % 26 + 97) }
        else { $_ }
    })
}

# ------------------------------------------------------------
#  Artifact collection (run history, last-opened, etc.)
# ------------------------------------------------------------
function Add-Record {
    param([string]$Path, [string]$Kind, $Time, [int]$Runs = 0, [string]$Source)
    if (-not $Path) { return }
    $Key = $Path.ToLower()
    if (-not $Script:Records.ContainsKey($Key)) {
        $Script:Records[$Key] = [pscustomobject]@{
            Path = $Path; LastRun = $null; LastPf = $null; LastOpen = $null; Runs = 0
            Sources = (New-Object System.Collections.Generic.List[string])
        }
    }
    $R = $Script:Records[$Key]
    switch ($Kind) {
        'Run'  { if ($Time -and (-not $R.LastRun  -or $Time -gt $R.LastRun))  { $R.LastRun  = $Time }; if ($Runs -gt $R.Runs) { $R.Runs = $Runs } }
        'Pf'   { if ($Time -and (-not $R.LastPf   -or $Time -gt $R.LastPf))   { $R.LastPf   = $Time } }
        'Open' { if ($Time -and (-not $R.LastOpen -or $Time -gt $R.LastOpen)) { $R.LastOpen = $Time } }
    }
    if ($Source -and -not $R.Sources.Contains($Source)) { $R.Sources.Add($Source) }
}

function Get-ArtifactRecords {
    $Script:Records      = @{}
    $Script:ConnectedUsb = @(Get-UsbVolumes | ForEach-Object { $_.DeviceID.ToUpper() })
    $Internal = Get-InternalLetters
    $VolMap   = Get-VolumeMap

    # 1) Recent .lnk files -> last opened (target path survives after the file is deleted)
    try {
        $Shell  = New-Object -ComObject WScript.Shell
        $Recent = "$env:APPDATA\Microsoft\Windows\Recent"
        foreach ($Link in [System.IO.Directory]::GetFiles($Recent, "*.lnk")) {
            try {
                $Target = $Shell.CreateShortcut($Link).TargetPath
                if (Test-ExternalPath $Target $Internal) {
                    Add-Record -Path $Target -Kind 'Open' -Time ((Get-Item $Link -Force).LastWriteTime) -Source 'Recent'
                }
            } catch {}
        }
    } catch {}

    # 2) UserAssist -> last execution time + run count (ROT13 encoded, per user)
    try {
        $UA = 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\UserAssist'
        foreach ($Guid in (Get-ChildItem $UA -ErrorAction SilentlyContinue)) {
            $CountKey = Get-Item "$($Guid.PSPath)\Count" -ErrorAction SilentlyContinue
            if (-not $CountKey) { continue }
            foreach ($Name in $CountKey.GetValueNames()) {
                $Path = Invoke-Rot13 $Name
                if (-not (Test-ExternalPath $Path $Internal)) { continue }
                $D = $CountKey.GetValue($Name)
                if ($D -is [byte[]] -and $D.Length -ge 68) {
                    $Runs = [int][BitConverter]::ToUInt32($D, 4)
                    $Ft   = [BitConverter]::ToInt64($D, 60)
                    $When = if ($Ft -gt 0) { [DateTime]::FromFileTime($Ft) } else { $null }
                    Add-Record -Path $Path -Kind 'Run' -Time $When -Runs $Runs -Source 'UserAssist'
                }
            }
        }
    } catch {}

    # 3) BAM -> last execution time (needs admin; device paths so unmounted volumes show as [HarddiskVolumeN])
    try {
        $BamRoots = @(
            'HKLM:\SYSTEM\CurrentControlSet\Services\bam\State\UserSettings',
            'HKLM:\SYSTEM\CurrentControlSet\Services\bam\UserSettings'
        )
        foreach ($Root in $BamRoots) {
            foreach ($SidKey in (Get-ChildItem $Root -ErrorAction SilentlyContinue)) {
                foreach ($Name in $SidKey.GetValueNames()) {
                    if ($Name -notmatch '^\\Device\\HarddiskVolume(\d+)(\\.*)$') { continue }
                    $Num  = $Matches[1]
                    $Rest = $Matches[2]
                    $Vol  = "\Device\HarddiskVolume$Num"
                    if ($VolMap.ContainsKey($Vol)) {
                        $Path = $VolMap[$Vol] + $Rest
                        if (-not (Test-ExternalPath $Path $Internal)) { continue }
                    } else {
                        $Path = "[HarddiskVolume$Num]$Rest"
                    }
                    $D = $SidKey.GetValue($Name)
                    if ($D -is [byte[]] -and $D.Length -ge 8) {
                        $Ft = [BitConverter]::ToInt64($D, 0)
                        if ($Ft -gt 0) { Add-Record -Path $Path -Kind 'Run' -Time ([DateTime]::FromFileTime($Ft)) -Source 'BAM' }
                    }
                }
            }
        }
    } catch {}

    # 4) Prefetch -> matched by exe NAME only (kept separate because it can be a different program with the same name)
    try {
        $Pf = @(Get-ChildItem 'C:\Windows\Prefetch' -Filter '*.pf' -ErrorAction SilentlyContinue)
        if ($Pf.Count -gt 0) {
            foreach ($R in @($Script:Records.Values)) {
                if ($R.Path -notmatch '\.exe$') { continue }
                $Exe = [System.IO.Path]::GetFileName($R.Path).ToUpper()
                $Hit = $Pf | Where-Object { $_.Name -like "$Exe-*.pf" } | Sort-Object LastWriteTime -Descending | Select-Object -First 1
                if ($Hit) { Add-Record -Path $R.Path -Kind 'Pf' -Time $Hit.LastWriteTime -Source 'Prefetch*' }
            }
        }
    } catch {}
}

function Get-PathStatus {
    param([string]$Path)
    if ($Path -match '^([A-Za-z]:)\\' -and $Script:ConnectedUsb -contains $Matches[1].ToUpper()) {
        if (Test-Path -LiteralPath $Path) { return 'ON DRIVE' } else { return 'DELETED/MISSING' }
    }
    return 'drive not connected'
}

function Get-RecordNote {
    param($R, [switch]$NoStatus)
    $Bits = @()
    if ($R.LastRun)  { $Bits += ("ran: " + (FormatTime $R.LastRun) + $(if ($R.Runs -gt 0) { " x$($R.Runs)" } else { '' })) }
    if ($R.LastPf)   { $Bits += ("prefetch*: " + (FormatTime $R.LastPf)) }
    if ($R.LastOpen) { $Bits += ("opened: " + (FormatTime $R.LastOpen)) }
    if (-not $NoStatus) { $Bits += (Get-PathStatus $R.Path) }
    if ($Bits.Count -gt 0) { return '[' + ($Bits -join ' | ') + ']' }
    return ''
}

# ------------------------------------------------------------
#  Device history (every USB storage device ever plugged in)
# ------------------------------------------------------------
function Get-UsbDeviceHistory {
    $Rows = @()
    $Devs = @(Get-PnpDevice -Class DiskDrive -ErrorAction SilentlyContinue | Where-Object { $_.InstanceId -like 'USBSTOR\*' })

    foreach ($D in $Devs) {
        $P = @{}
        foreach ($K in 'DEVPKEY_Device_FirstInstallDate', 'DEVPKEY_Device_LastArrivalDate', 'DEVPKEY_Device_LastRemovalDate') {
            $V = Get-PnpDeviceProperty -InstanceId $D.InstanceId -KeyName $K -ErrorAction SilentlyContinue
            if ($V -and $V.Data) { $P[$K] = $V.Data }
        }
        $Rows += [pscustomobject]@{
            Name    = $D.FriendlyName
            Serial  = (($D.InstanceId -split '\\')[-1] -replace '&\d+$', '')
            First   = $P['DEVPKEY_Device_FirstInstallDate']
            Arrival = $P['DEVPKEY_Device_LastArrivalDate']
            Removal = $P['DEVPKEY_Device_LastRemovalDate']
            Status  = if ($D.Status -eq 'OK') { 'CONNECTED NOW' } else { 'not connected' }
        }
    }

    if ($Rows.Count -eq 0) {
        # Fallback: registry only (no timestamps)
        $Base = 'HKLM:\SYSTEM\CurrentControlSet\Enum\USBSTOR'
        foreach ($Class in (Get-ChildItem $Base -ErrorAction SilentlyContinue)) {
            foreach ($Inst in (Get-ChildItem $Class.PSPath -ErrorAction SilentlyContinue)) {
                $Props = Get-ItemProperty $Inst.PSPath -ErrorAction SilentlyContinue
                $Rows += [pscustomobject]@{
                    Name = if ($Props.FriendlyName) { $Props.FriendlyName } else { $Class.PSChildName }
                    Serial = ($Inst.PSChildName -replace '&\d+$', '')
                    First = $null; Arrival = $null; Removal = $null; Status = 'unknown'
                }
            }
        }
    }

    $Rows | Sort-Object { if ($_.Arrival) { $_.Arrival } else { [datetime]::MinValue } } -Descending | ForEach-Object {
        [pscustomobject]@{
            'Device'          = $_.Name
            'Serial'          = $_.Serial
            'First Connected' = FormatTime $_.First
            'Last Connected'  = FormatTime $_.Arrival
            'Last Removed'    = FormatTime $_.Removal
            'Status'          = $_.Status
        }
    }
}

# ------------------------------------------------------------
#  Tree builder (for history - built from artifacts, not from the live drive)
# ------------------------------------------------------------
function New-TreeNode {
    @{
        Children = (New-Object 'System.Collections.Generic.SortedDictionary[string,object]' -ArgumentList ([StringComparer]::OrdinalIgnoreCase))
        Note     = ''
    }
}

function Add-TreePath {
    param($Node, [string[]]$Parts, [string]$Note)
    foreach ($P in $Parts) {
        if (-not $Node.Children.ContainsKey($P)) { $Node.Children[$P] = New-TreeNode }
        $Node = $Node.Children[$P]
    }
    $Node.Note = $Note
}

function Write-TreeNode {
    param($Node, [string]$Prefix, $Lines)
    $Keys = @($Node.Children.Keys)
    for ($i = 0; $i -lt $Keys.Count; $i++) {
        $K      = $Keys[$i]
        $IsLast = ($i -eq $Keys.Count - 1)
        $Child  = $Node.Children[$K]
        $Branch = if ($IsLast) { '\---' } else { '+---' }
        $Text   = if ($Child.Note) { "$K   $($Child.Note)" } else { $K }
        $Lines.Add("$Prefix$Branch$Text")
        $NextPrefix = if ($IsLast) { "$Prefix    " } else { "$Prefix|   " }
        Write-TreeNode $Child $NextPrefix $Lines
    }
}

# ------------------------------------------------------------
#  1 -> 1 : CHECK HISTORY
# ------------------------------------------------------------
function RunUsbHistory {
    Clear-Host
    Write-Host ">>> Building history from leftover artifacts..." -ForegroundColor Green
    if (-not $Script:IsAdmin) {
        Write-Host "[!] Not Administrator: BAM, Prefetch and some device timestamps will be missing." -ForegroundColor Yellow
    }
    Get-ArtifactRecords

    $Data = New-Object System.Collections.Generic.List[string]
    Emit $Data "=== USB HISTORY REPORT ===" Cyan
    Emit $Data "Computer: $env:COMPUTERNAME    User: $env:USERNAME    Admin: $Script:IsAdmin" Gray
    Emit $Data "Generated: $(FormatTime (Get-Date))" Gray

    # [1] devices
    Emit $Data "" Gray
    Emit $Data "=== [1] USB STORAGE DEVICES EVER CONNECTED ===" Cyan
    $Devs = @(Get-UsbDeviceHistory)
    if ($Devs.Count -gt 0) {
        Emit $Data (($Devs | Format-Table -AutoSize | Out-String -Width 4096).TrimEnd()) White
    } else {
        Emit $Data "(no USB storage devices found)" DarkGray
    }

    # [2] last run times
    Emit $Data "" Gray
    Emit $Data "=== [2] FILES RUN FROM EXTERNAL DRIVES (last run) ===" Yellow
    $Ran = @($Script:Records.Values | Where-Object { $_.LastRun -or $_.LastPf } |
        Sort-Object { if ($_.LastRun) { $_.LastRun } else { $_.LastPf } } -Descending)
    if ($Ran.Count -gt 0) {
        $Table = $Ran | ForEach-Object {
            [pscustomobject]@{
                'Last Run'   = FormatTime $_.LastRun
                'Runs'       = if ($_.Runs -gt 0) { $_.Runs } else { '-' }
                'Prefetch*'  = FormatTime $_.LastPf
                'Status'     = Get-PathStatus $_.Path
                'Path'       = $_.Path
                'Source'     = ($_.Sources -join ',')
            }
        }
        Emit $Data (($Table | Format-Table -AutoSize | Out-String -Width 4096).TrimEnd()) White
    } else {
        Emit $Data "(no execution records for external drives found)" DarkGray
    }

    # [3] opened only
    Emit $Data "" Gray
    Emit $Data "=== [3] FILES OPENED FROM EXTERNAL DRIVES (no run record) ===" Yellow
    $Opened = @($Script:Records.Values | Where-Object { $_.LastOpen -and -not $_.LastRun -and -not $_.LastPf } |
        Sort-Object LastOpen -Descending)
    if ($Opened.Count -gt 0) {
        $Table = $Opened | ForEach-Object {
            [pscustomobject]@{
                'Last Opened' = FormatTime $_.LastOpen
                'Status'      = Get-PathStatus $_.Path
                'Path'        = $_.Path
            }
        }
        Emit $Data (($Table | Format-Table -AutoSize | Out-String -Width 4096).TrimEnd()) White
    } else {
        Emit $Data "(none)" DarkGray
    }

    # [4] reconstructed tree
    Emit $Data "" Gray
    Emit $Data "=== [4] RECONSTRUCTED FILE TREE (from artifacts, includes deleted files) ===" Cyan
    if ($Script:Records.Count -gt 0) {
        $Groups = $Script:Records.Values | Group-Object { if ($_.Path -match '^([^\\]+)\\') { $Matches[1].ToUpper() } else { $_.Path } }
        foreach ($G in ($Groups | Sort-Object Name)) {
            $Root = New-TreeNode
            foreach ($R in $G.Group) {
                $Rest = ''
                if ($R.Path -match '^([^\\]+)\\(.*)$') { $Rest = $Matches[2] }
                $Parts = @($Rest -split '\\' | Where-Object { $_ })
                if ($Parts.Count -gt 0) { Add-TreePath $Root $Parts (Get-RecordNote $R) }
            }
            $Lines = New-Object System.Collections.Generic.List[string]
            Write-TreeNode $Root '' $Lines
            $Connected = if ($Script:ConnectedUsb -contains $G.Name) { '(connected now)' } else { '(not connected)' }
            Emit $Data "" Gray
            Emit $Data "$($G.Name)\  $Connected" Cyan
            Emit $Data ($Lines -join "`r`n") Cyan
        }
    } else {
        Emit $Data "(nothing to reconstruct)" DarkGray
    }

    Emit $Data "" Gray
    Emit $Data "--- Notes ---" DarkGray
    Emit $Data "ran = UserAssist/BAM last execution time. prefetch* = matched by exe NAME only (could be a different program with the same name)." DarkGray
    Emit $Data "opened = Recent .lnk last-write time. [HarddiskVolumeN] = a drive that is no longer connected (BAM only knows the volume number)." DarkGray
    Emit $Data "Drive letters get reused, so old entries for a letter may belong to a different USB." DarkGray

    PushToDiscord -Content ($Data -join "`r`n") -Name "USB History Report.txt" -CustomHeader "# $(AliasUpper)'s USB History Report"
    ReturnToMenu
}

# ------------------------------------------------------------
#  1 -> 2 : CHECK CURRENT
# ------------------------------------------------------------
function RunUsbCurrent {
    Clear-Host
    Write-Host ">>> Scanning system for live hardware..." -ForegroundColor Green
    $Drives = @(Get-UsbVolumes)

    if ($Drives.Count -eq 0) {
        Write-Host "No USB drive detected. Plug one in while the script is open first." -ForegroundColor Red
        ReturnToMenu
        return
    }

    Get-ArtifactRecords

    Write-Host "Active hardware discovered. Processing connected media..." -ForegroundColor Green
    Get-PnpDevice -Class DiskDrive -Status OK -ErrorAction SilentlyContinue |
        Select-Object Status, Class, FriendlyName, InstanceId | Format-Table -AutoSize

    foreach ($Drive in $Drives) {
        $Letter = $Drive.DeviceID
        $Size   = if ($Drive.Size) { "$([math]::Round($Drive.Size / 1GB, 1)) GB" } else { "unknown" }
        $Data   = New-Object System.Collections.Generic.List[string]

        Emit $Data "Drive: $Letter   Label: $($Drive.VolumeName)   FS: $($Drive.FileSystem)   Size: $Size   Serial: $($Drive.VolumeSerialNumber)" Green
        Emit $Data "" Gray

        $Files = @(Get-ChildItem -LiteralPath "$Letter\" -Recurse -File -Force -ErrorAction SilentlyContinue)
        $RunItems  = @()
        $IdleItems = @()
        foreach ($File in $Files) {
            $Rec  = $Script:Records[$File.FullName.ToLower()]
            $Note = if ($Rec) { Get-RecordNote $Rec -NoStatus } else { '' }
            $Line = if ($Note) { "$($File.FullName)   $Note" } else { $File.FullName }
            $Touched = $Rec -and (
                ($Rec.LastRun  -and $Rec.LastRun  -ge $Script:SessionStartTime) -or
                ($Rec.LastOpen -and $Rec.LastOpen -ge $Script:SessionStartTime))
            if ($Touched) { $RunItems += $Line } else { $IdleItems += $Line }
        }

        Emit $Data "In USB:" Gray
        if ($IdleItems.Count -gt 0) { foreach ($I in $IdleItems) { Emit $Data $I Gray } }
        else { Emit $Data "(No untouched files found)" DarkGray }

        Emit $Data "-------------------------" Gray
        Emit $Data "Ran from USB:" Yellow
        if ($RunItems.Count -gt 0) { foreach ($I in $RunItems) { Emit $Data "[LIVE INTERACTION] -> $I" Yellow } }
        else { Emit $Data "(No executed files found during this session)" DarkGray }

        Emit $Data "-------------------------" Gray
        Emit $Data "Previously seen on this drive, no longer present:" Magenta
        $Gone = @($Script:Records.Values | Where-Object {
            $_.Path -like "$Letter\*" -and -not (Test-Path -LiteralPath $_.Path)
        } | Sort-Object Path)
        if ($Gone.Count -gt 0) { foreach ($R in $Gone) { Emit $Data "$($R.Path)   $(Get-RecordNote $R -NoStatus)" Magenta } }
        else { Emit $Data "(none)" DarkGray }

        Emit $Data "-------------------------" Gray
        Emit $Data "Latest use:" Gray
        Emit $Data "Snapshot Captured At: $((Get-Date).ToString('yyyy-MM-dd HH:mm:ss'))" Gray
        Emit $Data "=========================================================" Gray

        $Tree = tree ("$Letter\") /A /F
        if ($Tree) { foreach ($T in $Tree) { Emit $Data $T Cyan } }

        PushToDiscord -Content ($Data -join "`r`n") -Name "USB Activity Report.txt" -CustomHeader "# $(AliasUpper)'s USB Activity Report"
    }
    ReturnToMenu
}

function RunUsbMenu {
    while ($true) {
        Clear-Host
        Write-Output ""
        Write-Output "  [ USB ACTIVITY ]"
        Write-Output "  ------------------------------------------"
        Write-Output "  1. Check History"
        Write-Output "  2. Check Current"
        Write-Output "  B. Back"
        Write-Output ""
        $Sub = (Read-Host "Choose option").ToUpper()
        switch ($Sub) {
            '1' { RunUsbHistory }
            '2' { RunUsbCurrent }
            'B' { return }
            Default { Write-Host "Invalid selection. Try again." -ForegroundColor Red; Start-Sleep 1 }
        }
    }
}

# ------------------------------------------------------------
#  2 : VIEW USBs
# ------------------------------------------------------------
function RunRegistryCheck {
    Clear-Host
    $Data = New-Object System.Collections.Generic.List[string]

    Emit $Data ">>> USB Devices (connection times)" Magenta
    Emit $Data "----------------------------------------------------" Gray
    $Devs = @(Get-UsbDeviceHistory)
    if ($Devs.Count -gt 0) { Emit $Data (($Devs | Format-Table -AutoSize | Out-String -Width 4096).TrimEnd()) White }
    else { Emit $Data "No USB storage devices found." DarkGray }

    Emit $Data "" Gray
    Emit $Data ">>> Hardware Installation History (SetupAPI Log, USB only)" Magenta
    Emit $Data "----------------------------------------------------" Gray
    $SetupLog = "C:\Windows\INF\setupapi.dev.log"
    if (Test-Path $SetupLog) {
        $Hits = @(Select-String -Path $SetupLog -Pattern '>>>  \[Device Install' -Context 0, 1 -ErrorAction SilentlyContinue |
            Where-Object { $_.Line -match 'USB' })
        if ($Hits.Count -gt 0) {
            foreach ($H in ($Hits | Select-Object -Last $LogCount)) {
                $Dev  = ($H.Line -replace '^>>>\s+\[Device Install \([^)]*\) - ', '' -replace '\]\s*$', '')
                $When = if ($H.Context.PostContext.Count -gt 0) { ($H.Context.PostContext[0] -replace '^>>>\s+Section start\s+', '').Trim() } else { '' }
                Emit $Data "$When   $Dev" Gray
            }
        } else {
            Emit $Data "No relevant USB device setup entries found in log." DarkGray
        }
    } else {
        Emit $Data "SetupAPI log file is missing or inaccessible." Red
    }

    PushToDiscord -Content ($Data -join "`r`n") -Name "USB Devices Report.txt" -CustomHeader "# $(AliasUpper)'s USB History Report"
    ReturnToMenu
}

# ------------------------------------------------------------
#  3 : POWERSHELL HISTORY
# ------------------------------------------------------------
function RunPowerShellAudit {
    Clear-Host
    Write-Output "=== Powershell History ==="
    Write-Output "1. Past 1 Hour"
    Write-Output "2. Past 12 Hours"
    Write-Output "3. Past 24 Hours"
    Write-Output "4. Past 48 Hours"
    Write-Output ""
    $Selection = Read-Host "Select timeframe option"

    if     ($Selection -eq '1') { $Cutoff = (Get-Date).AddHours(-1);  $HoursLabel = "1 Hour" }
    elseif ($Selection -eq '2') { $Cutoff = (Get-Date).AddHours(-12); $HoursLabel = "12 Hours" }
    elseif ($Selection -eq '3') { $Cutoff = (Get-Date).AddHours(-24); $HoursLabel = "24 Hours" }
    elseif ($Selection -eq '4') { $Cutoff = (Get-Date).AddHours(-48); $HoursLabel = "48 Hours" }
    else {
        Write-Host "Invalid selection. Aborting." -ForegroundColor Red
        Start-Sleep 1
        return
    }

    Clear-Host
    $Data = New-Object System.Collections.Generic.List[string]
    Emit $Data ">>> Unfiltered PowerShell Script Execution Log" Cyan
    Emit $Data "Timeframe Window Triggered: $Cutoff" Gray
    Emit $Data "----------------------------------------------------" Gray

    $Events = Get-WinEvent -FilterHashtable @{
        LogName   = 'Microsoft-Windows-PowerShell/Operational'
        ID        = 4104
        StartTime = $Cutoff
    } -ErrorAction SilentlyContinue

    if ($Events) {
        $Deduper = @{}
        foreach ($Ev in $Events) {
            $ScriptText = [string]$Ev.Properties[2].Value
            if (-not [string]::IsNullOrWhiteSpace($ScriptText)) {
                if (-not $Deduper.ContainsKey($ScriptText)) {
                    $Deduper[$ScriptText] = New-Object System.Collections.Generic.List[string]
                }
                $Deduper[$ScriptText].Add($Ev.TimeCreated.ToString("yyyy-MM-dd HH:mm:ss"))
            }
        }

        foreach ($Key in $Deduper.Keys) {
            $Timestamps = $Deduper[$Key]
            $Count = $Timestamps.Count
            $ScriptHeader = if ($Count -gt 1) { "$Key (x$Count ($($Timestamps -join ', ')))" }
                            else { "[$($Timestamps[0])] $Key" }
            Emit $Data $ScriptHeader White
            Emit $Data "----------------------------------------------------" Gray
            $Data.Add("`r`n`r`n")
        }
    } else {
        Emit $Data "No recorded PowerShell execution block log artifacts discovered in this timeframe." DarkGray
    }

    Emit $Data "" Gray
    Emit $Data ">>> Active Memory Monitoring Process State Scan" Green
    Emit $Data "----------------------------------------------------" Gray

    $Processes = Get-CimInstance Win32_Process -Filter "Name like 'powershell%'" |
                 Where-Object { $_.ProcessId -ne $PID -and $_.CommandLine -notmatch 'ft -auto' }

    if ($Processes) {
        Emit $Data ($Processes | Select-Object ProcessName, ProcessId, CommandLine | Format-Table -AutoSize | Out-String -Width 4096).TrimEnd() White
    } else {
        Emit $Data "No operational background PowerShell instances located in active execution space." DarkGray
    }

    PushToDiscord -Content ($Data -join "`r`n") -Name "$HoursLabel Powershell Report.txt" -CustomHeader "# $(AliasUpper)'s $HoursLabel Powershell History Report"
    ReturnToMenu
}
